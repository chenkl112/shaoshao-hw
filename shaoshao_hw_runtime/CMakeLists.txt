  shaoshao_hw_interfaces hw_omni_controller)
function(add_runtime_node target source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE include)
  target_link_libraries(${target} Eigen3::Eigen yaml-cpp OpenSSL::Crypto JsonCpp::JsonCpp PNG::PNG ${MINIZIP_LIBRARY} ZLIB::ZLIB)
  ament_target_dependencies(${target} ${RUNTIME_DEPS})
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endfunction()

add_runtime_node(index_recorder src/index_recorder.cpp)
