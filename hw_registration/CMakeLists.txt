cmake_minimum_required(VERSION 3.16)
project(hw_registration)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(shaoshao_hw_interfaces REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_path(SMALL_GICP_INCLUDE small_gicp/registration/registration.hpp
  HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/small_gicp/include" REQUIRED)
add_executable(registration_server src/registration_server.cpp)
target_include_directories(registration_server PRIVATE ${SMALL_GICP_INCLUDE} ${PCL_INCLUDE_DIRS})
target_link_libraries(registration_server Eigen3::Eigen OpenMP::OpenMP_CXX ${PCL_LIBRARIES})
ament_target_dependencies(registration_server rclcpp sensor_msgs geometry_msgs pcl_conversions shaoshao_hw_interfaces)
install(TARGETS registration_server DESTINATION lib/${PROJECT_NAME})
add_library(shaoshao_norm_critic SHARED src/norm_critic.cpp)
ament_target_dependencies(shaoshao_norm_critic nav2_mppi_controller pluginlib)
pluginlib_export_plugin_description_file(nav2_mppi_controller norm_critic.xml)
install(TARGETS shaoshao_norm_critic DESTINATION lib)
ament_package()
