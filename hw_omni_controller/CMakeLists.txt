cmake_minimum_required(VERSION 3.16)
project(hw_omni_controller)
set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(shaoshao_hw_interfaces REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io kdtree)
add_library(hw_omni_controller SHARED src/controller.cpp)
target_include_directories(hw_omni_controller PRIVATE include vendor)
target_link_libraries(hw_omni_controller Eigen3::Eigen)
target_compile_options(hw_omni_controller PRIVATE -Wall -Wextra)
ament_target_dependencies(hw_omni_controller rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav_msgs std_msgs
  shaoshao_hw_interfaces tf2_geometry_msgs tf2_ros pluginlib)
pluginlib_export_plugin_description_file(nav2_core controller.xml)
install(TARGETS hw_omni_controller DESTINATION lib)
add_executable(obstacle_predictor src/obstacle_predictor.cpp)
target_include_directories(obstacle_predictor PRIVATE include ${PCL_INCLUDE_DIRS})
target_link_libraries(obstacle_predictor Eigen3::Eigen ${PCL_LIBRARIES})
ament_target_dependencies(obstacle_predictor rclcpp sensor_msgs std_msgs shaoshao_hw_interfaces tf2_ros)
add_executable(command_relay src/command_relay.cpp)
target_include_directories(command_relay PRIVATE include)
target_link_libraries(command_relay Eigen3::Eigen)
ament_target_dependencies(command_relay rclcpp geometry_msgs nav_msgs std_msgs shaoshao_hw_interfaces tf2_ros tf2_geometry_msgs)
install(TARGETS obstacle_predictor command_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY vendor/ DESTINATION share/${PROJECT_NAME}/vendor)
install(DIRECTORY include/ DESTINATION include)
install(FILES vendor/fddp_solver.hpp DESTINATION include/hw_omni_controller)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3)
ament_package()
