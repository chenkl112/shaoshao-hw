cmake_minimum_required(VERSION 3.16)
project(shaoshao_web LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SHAOSHAO_WEB_ROS "Build live ROS telemetry" ON)
# Ubuntu Drogon 1.8 only searches obsolete mysqlclient_r/mariadbclient names.
# Resolve the standard MySQL client first so PCL and Drogon can coexist.
find_library(MYSQL_LIBRARIES NAMES mysqlclient_r mariadbclient mysqlclient)
find_package(Drogon CONFIG REQUIRED)
add_executable(shaoshao_web src/main.cpp)
target_include_directories(shaoshao_web PRIVATE include)
target_link_libraries(shaoshao_web PRIVATE Drogon::Drogon)
if(SHAOSHAO_WEB_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(livox_ros_driver2 REQUIRED)
  find_package(hw_omni_controller REQUIRED)
  find_package(Eigen3 REQUIRED)
  target_sources(shaoshao_web PRIVATE src/telemetry.cpp)
  target_compile_definitions(shaoshao_web PRIVATE SHAOSHAO_WEB_ROS)
  target_link_libraries(shaoshao_web PRIVATE Eigen3::Eigen)
  ament_target_dependencies(shaoshao_web PUBLIC rclcpp sensor_msgs std_msgs livox_ros_driver2 hw_omni_controller)
  install(TARGETS shaoshao_web DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
